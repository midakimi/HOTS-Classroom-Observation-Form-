const SHEET_SUPERVISION = "Supervision"; const SHEET_TEACHERS = "Teachers"; const SHEET_APP_DATA = "AppData"; function getOrCreateSheet(ss, name, headers) { let sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); if (headers && headers.length) sheet.appendRow(headers); } return sheet; } function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const supSheet = ss.getSheetByName(SHEET_SUPERVISION); const tchSheet = ss.getSheetByName(SHEET_TEACHERS); const appSheet = getOrCreateSheet(ss, SHEET_APP_DATA, ["collection", "id", "json", "updatedAt"]); return jsonResponse({ status: "success", supervision: getSheetData(supSheet), teachers: getSheetData(tchSheet), appData: getSheetData(appSheet) }); } function doPost(e) { try { const payload = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); const action = payload.action || "save"; const colName = payload.collection; const id = payload.id; const record = payload.record; // 1. บันทึกก้อนข้อมูลกลาง (เพื่อให้หน้าเว็บซิงก์ครบ 100% ทุกแท็บ) const appSheet = getOrCreateSheet(ss, SHEET_APP_DATA, ["collection", "id", "json", "updatedAt"]); upsertOrDeleteAppData(appSheet, action, colName, id, record); // 2. ซิงก์ลงตารางอ่านง่าย (Supervision และ Teachers) if (colName === "obs") { const supSheet = ss.getSheetByName(SHEET_SUPERVISION); if (supSheet) syncToReadableSheet(supSheet, action, id, mapObsToRow(record)); } else if (colName === "roster") { const tchSheet = ss.getSheetByName(SHEET_TEACHERS); if (tchSheet) syncToReadableSheet(tchSheet, action, id, mapTeacherToRow(record)); } return jsonResponse({ status: "success", message: "ซิงก์ออนไลน์เรียบร้อย" }); } catch (error) { return jsonResponse({ status: "error", message: error.toString() }); } } function upsertOrDeleteAppData(sheet, action, colName, id, record) { const data = sheet.getDataRange().getValues(); let foundRow = -1; for (let i = 1; i < data.length; i++) { if (String(data[i]) === String(colName) && String(data[i]) === String(id)) { foundRow = i + 1; break; } } if (action === "delete") { if (foundRow > 0) sheet.deleteRow(foundRow); } else { const rowValues = [colName, id, JSON.stringify(record), new Date().toLocaleString("th-TH")]; if (foundRow > 0) sheet.getRange(foundRow, 1, 1, 4).setValues([rowValues]); else sheet.appendRow(rowValues); } } function syncToReadableSheet(sheet, action, id, mappedObj) { const data = sheet.getDataRange().getValues(); if (data.length === 0) return; const headers = data; const idColIndex = headers.findIndex(h => String(h).toLowerCase().includes("id")); let foundRow = -1; if (idColIndex >= 0) { for (let i = 1; i < data.length; i++) { if (String(data[i][idColIndex]) === String(id)) { foundRow = i + 1; break; } } } if (action === "delete") { if (foundRow > 0) sheet.deleteRow(foundRow); return; } const rowValues = headers.map(h => { const key = String(h).trim(); return mappedObj[key] !== undefined ? mappedObj[key] : ""; }); if (foundRow > 0) { sheet.getRange(foundRow, 1, 1, rowValues.length).setValues([rowValues]); } else { sheet.appendRow(rowValues); } } function mapObsToRow(r) { if (!r) return {}; return { "ID": r.id || "", "Date": r.date || "", "Supervisor": r.observer || "", "Term": r.cyc || "", "Teacher": r.teacher || "", "School": r.school || "", "Subject": r.subject || "", "Grade": r.grade || "", "Topic": r.topic || "", "Phase": r.phase || "", "Student_Count": (r.ctx && r.ctx.n) || "", "Resources": (r.ctx && r.ctx.res) || "", "Teaching_load": (r.ctx && r.ctx.load) || "", "Limitations": (r.ctx && r.ctx.lim) || "", "Scores": JSON.stringify(r.r || {}), "Average_score": r._avg || "", "Hots_percent": r._hq || "", "Feedback": [r.str, r.imp, r.act].filter(Boolean).join(" | "), "Timestamp": new Date().toLocaleString("th-TH") }; } function mapTeacherToRow(t) { if (!t) return {}; return { "teacher_id": t.id || "", "prefix": "", "fullname": t.name || "", "school": t.school || "", "subject": t.subject || "", "grade": t.grade || "" }; } function getSheetData(sheet) { if (!sheet) return []; const data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; const headers = data; return data.slice(1).map(row => { let obj = {}; headers.forEach((h, i) => obj[h] = row[i]); return obj; }); } function jsonResponse(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
